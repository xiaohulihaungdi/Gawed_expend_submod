# ------------------------------------------------------------
# .1  加韦德支持克罗维斯（原 flavor_grombar.50 选项 a 的效果）
# ------------------------------------------------------------
country_event = {
	id = gawed_yongdongbianqu.1
	title = flavor_grombar.50.t
	desc = flavor_grombar.50.d
	picture = CARIBBEAN_PIRATE_FORT_eventPicture

	fire_only_once = yes
	is_triggered_only = yes

	option = {
		name = flavor_grombar.50.a
		ai_chance = { factor = 100 }

		if = {
			limit = { has_saved_event_target = crovis_spawn_tag }
			create_alliance = event_target:crovis_spawn_tag
			add_historical_friend = event_target:crovis_spawn_tag
			event_target:crovis_spawn_tag = { add_historical_friend = ROOT }
			event_target:crovis_spawn_tag = {
				country_event = {
					id = flavor_grombar.52
					days = 1
				}
			}
		}
		else = {
			Z26 = { create_alliance = ROOT }
		}

		add_prestige = 10
	}
}

	}

	option = { name = gawed_yongdongbianqu.3.a }
}

country_event = {

		738 = { change_culture = white_reachman }
		739 = { change_culture = white_reachman }
		740 = { change_culture = white_reachman }
		990 = { change_culture = white_reachman }

		country_event = {
			id = gawed_yongdongbianqu.5
			days = 90
		}
	}
}


# ------------------------------------------------------------
# .5  永冻边区的归属之争（由 .4 触发，90 天后到来）
#     选项 a：让阿德肖（Z26）与永冻边区（Z45）独立 → 1 天后触发 .6
#     选项 b：两国独立倾向 -10%，并获得 10 年修正 bumanderenmin
# ------------------------------------------------------------
country_event = {
	id = gawed_yongdongbianqu.5
	title = gawed_yongdongbianqu.5.t
	desc = gawed_yongdongbianqu.5.d
	picture = ORNATE_THRONE_ROOM_eventPicture

	fire_only_once = yes
	is_triggered_only = yes

	# 选项 a：放他们独立，随后翻脸
	option = {
		name = gawed_yongdongbianqu.5.a
		ai_chance = { factor = 50 }

		if = {
			limit = {
				exists = Z26
				Z26 = { is_subject = yes }
			}
			Z26 = { grant_independence = yes }
		}

		if = {
			limit = {
				exists = Z45
				Z45 = { is_subject = yes }
			}
			Z45 = { grant_independence = yes }
		}

		add_prestige = -5

		# 1 天后触发第六个事件（两国对加韦德宣战）
		country_event = {
			id = gawed_yongdongbianqu.6
			days = 1
		}
	}

	# 选项 b：收紧控制
	option = {
		name = gawed_yongdongbianqu.5.b
		ai_chance = { factor = 50 }

		if = {
			limit = { exists = Z26 }
			Z26 = {
				add_liberty_desire = -10
				add_country_modifier = {
					name = bumanderenmin
					duration = 3650
				}
			}
		}

		if = {
			limit = { exists = Z45 }
			Z45 = {
				add_liberty_desire = -10
				add_country_modifier = {
					name = bumanderenmin
					duration = 3650
				}
			}
		}
	}
}

# .6  阿德肖与永冻边区对加韦德宣战（由 .5 选项 a 触发，1 天后）
# ------------------------------------------------------------

	desc = gawed_yongdongbianqu.6.d

	fire_only_once = yes
	is_triggered_only = yes

	option = {
		name = gawed_yongdongbianqu.6.a
		ai_chance = { factor = 100 }

		# 两国各自对加韦德（ROOT = A13）宣战
		# 想指定 CB 就把 declare_war 换成：

		if = {
			limit = {

				Z26 = {
					is_subject = no

				}
			}
			Z26 = { declare_war = ROOT }
		}

		if = {
			limit = {
				exists = Z45
				Z45 = {
					is_subject = no
					NOT = { war_with = ROOT }
					NOT = { truce_with = ROOT }
				}
			}
			Z45 = { declare_war = ROOT }
		}
	}
}
