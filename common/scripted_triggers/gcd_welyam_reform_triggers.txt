# 新增：始终核对专用 CB 和两位战争领袖，防止其他战争误触发吞并结算。
gcd_welyam_civil_war_active_trigger = {
    has_country_flag = gcd_welyam_civil_war_active
    is_in_war = {
        casus_belli = cb_gcd_gawedi_civil_war
        attacker_leader = A50
        defender_leader = A13
    }
}

# 新增：沿用任务原稿的两条完成路线，事件读取任务完成时保存的路线。
gcd_welyam_settlement_route_trigger = {
    estate_influence = { estate = estate_nobles influence = 50 }
    has_estate_privilege = estate_nobles_supremacy_over_crown
    treasury = 200
}
gcd_welyam_consolidation_route_trigger = {
    # 修正：第二路线严格低于40；正好40不满足，特权数量要求保留。
    NOT = { estate_influence = { estate = estate_nobles influence = 40 } }
    NOT = { num_of_estate_privileges = { estate = estate_nobles value = 2 } }
}
