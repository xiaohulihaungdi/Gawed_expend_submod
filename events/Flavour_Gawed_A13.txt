			random_list = {

					modifier = { # Relations should begin to sour a bit

						is_year = 1575
					}

				}
			}
		}
	}

		clr_country_flag = council_support_ruler
		clr_country_flag = council_oppose_ruler
	}

	option = {
		name = flavor_gawed.5.a

			factor = 3
		}
		trigger = {
			has_country_flag = council_support_ruler
		}

    		limit = {has_reform = empower_magnate_council}
			add_ruler_modifier = {
				name = magnate_council_support_empower
				duration = -1
			}
		}
		else = {
			add_ruler_modifier = {
				name = magnate_council_support

			}
		}
	}

			factor = 3
		}
		trigger = {
			has_country_flag = council_support_ruler
		}
		add_prestige = 10

			limit = {has_reform = empower_magnate_council}
			add_ruler_modifier = {
				name = magnate_council_rejected_empower

			}
		}
		else = {
