# 新增：从求援起安装独立每日检测，灾难结束不影响本剧情。
gcd_crovis_install_pulse_effect = {
    if = {
        limit = { NOT = { has_country_flag = gcd_crovis_pulse_installed } }
        set_country_flag = gcd_crovis_pulse_installed
        221 = { add_province_triggered_modifier = gcd_crovis_daily_pulse }
    }
}
gcd_crovis_stop_pulse_effect = {
    # 先关运行标记，再移除触发修正，防止移除回调重新执行。
    clr_country_flag = gcd_crovis_pulse_installed
    clr_country_flag = gcd_crovis_daily_tick
    221 = { remove_province_triggered_modifier = gcd_crovis_daily_pulse }
}

# 新增：在原版 .49/.54 运行国保存真实克洛维国家，跟随其复国改名。
gcd_crovis_record_actor_effect = {
    set_country_flag = gcd_crovis_actor
    save_global_event_target_as = gcd_crovis_actor
}

# 新增：原版开战回调直接通知玩家；每日入口兼容已开战/稍后承诺资助的存档。
# 只排队一次；未送达超过3天才补投。事件显示即记seen，停留选项不会重复弹窗。
gcd_crovis_queue_departure_effect = {
    if = {
        limit = {
            tag = A13
            gcd_crovis_departed_trigger = yes
            NOT = { has_country_flag = gcd_crovis_departure_seen }
            NOT = { has_country_flag = gcd_crovis_joined_restoration }
            OR = {
                NOT = { has_country_flag = gcd_crovis_departure_queued }
                had_country_flag = { flag = gcd_crovis_departure_queued days = 3 }
            }
        }
        clr_country_flag = gcd_crovis_departure_queued
        set_country_flag = gcd_crovis_departure_queued
        # 与原版事件栈分开，在开战后次日自动展示出征诗。
        country_event = { id = gawed_yongdongbianqu.1 days = 1 }
    }
}
gcd_crovis_on_restoration_war_effect = {
    gcd_crovis_record_actor_effect = yes
    A13 = { gcd_crovis_queue_departure_effect = yes }
}

gcd_crovis_join_restoration_effect = {
    if = {
        limit = { gcd_crovis_departed_trigger = yes }
        every_country = {
            limit = {
                war_with = Z18
                is_in_war = { casus_belli = cb_crovis_invasion attacker_leader = THIS defender_leader = Z18 }
            }
            set_country_flag = gcd_crovis_actor
            save_global_event_target_as = gcd_crovis_actor
        }
        # 与原版克洛维援军采用同一参战命令；原版剧情此时的进攻战争为复国战。
        join_all_offensive_wars_of = event_target:gcd_crovis_actor
        set_country_flag = gcd_crovis_joined_restoration
    }
}

# 调整：减伤改由任务发放；永久宣称按用户最新确认覆盖东艾伦河湾21省。
gcd_crovis_grant_task_reward_effect = {
    if = {
        limit = {
            gcd_crovis_in_restoration_war_trigger = yes
            NOT = { has_country_flag = gcd_crovis_task_reward_granted }
        }
        add_country_modifier = { name = gcd_crovis_funded_armies duration = -1 desc = gcd_crovis_buff_duration_tt }
        every_country = {
            limit = { is_in_war = { casus_belli = cb_crovis_invasion attacker_leader = THIS defender_leader = Z18 } }
            gcd_crovis_record_actor_effect = yes
            add_country_modifier = { name = gcd_crovis_funded_armies duration = -1 desc = gcd_crovis_buff_duration_tt }
        }
        every_province = {
            limit = { province_group = eastern_alenic_reach }
            add_permanent_claim = ROOT
        }
        set_country_flag = gcd_crovis_task_reward_granted
    }
}

# 新增：专属修正只留到复国战争结束，不沿用到阿德肖叛乱。
gcd_crovis_remove_support_effect = {
    A13 = { remove_country_modifier = gcd_crovis_funded_armies }
    every_country = {
        limit = { has_country_modifier = gcd_crovis_funded_armies }
        remove_country_modifier = gcd_crovis_funded_armies
    }
}

gcd_crovis_start_revolt_effect = {
    # 防护：必须是两国都存在的审判分支；不释放其他宗主的属国。
    if = {
        limit = { exists = Z45 exists = Z26 NOT = { has_country_flag = gcd_crovis_revolt_started } }
        set_country_flag = gcd_crovis_revolt_started
        Z45 = {
            if = { limit = { is_subject_of = A13 } grant_independence = yes }
            if = { limit = { alliance_with = A13 } break_alliance = A13 }
        }
        Z26 = {
            if = { limit = { is_subject_of = A13 } grant_independence = yes }
            if = { limit = { alliance_with = A13 } break_alliance = A13 }
        }
        Z45 = { declare_war_with_cb = { who = A13 casus_belli = cb_gcd_adshaw_revolt } }
        if = {
            limit = { is_in_war = { casus_belli = cb_gcd_adshaw_revolt attacker_leader = Z45 defender_leader = A13 } }
            set_country_flag = gcd_crovis_revolt_active
            Z26 = { create_alliance = Z45 join_all_offensive_wars_of = Z45 }
            # 转移我国及我国属国持有的西寒湾，保留中立国家的领土。
            west_chillsbay_area = {
                limit = { OR = { owned_by = A13 owned_by_subject_of = A13 } }
                cede_province = Z26
            }
            # 用户确认：这场战争就是原改革灾难机制预留的“阿德肖叛乱”。
            gcd_crovis_westmoors_hook_effect = yes
        }
    }
}

gcd_crovis_westmoors_hook_effect = {
    if = {
        limit = {
            gcd_crovis_revolt_active_trigger = yes
            has_disaster = gcd_welyam_reform
            # 新增：摩尔婚约永久豁免威斯特摩尔倒戈，不随国王死亡或灾难结束撤销。
            NOT = { has_country_flag = gcd_marriage_westmoors_loyal }
            A24 = { is_subject_of = A13 liberty_desire = 50.001 }
        }
        Z45 = { save_event_target_as = gcd_welyam_external_enemy }
        A24 = {
            grant_independence = yes
            if = { limit = { alliance_with = A13 } break_alliance = A13 }
            # 清除自动随宗主加入的防御方身份后，加入永冻边区进攻方。
            if = { limit = { war_with = Z45 } white_peace = Z45 }
            join_all_offensive_wars_of = Z45
        }
        country_event = { id = gcd_welyam.40 days = 1 }
    }
}

gcd_crovis_daily_effect = {
    if = {
        limit = {
            tag = A13 has_country_flag = gcd_crovis_pulse_installed
            OR = { NOT = { has_country_flag = gcd_crovis_daily_tick } had_country_flag = { flag = gcd_crovis_daily_tick days = 1 } }
        }
        set_country_flag = gcd_crovis_daily_tick
        gcd_crovis_queue_departure_effect = yes
        # 兼容：旧版任务已完成、仍在复国战中的测试存档补齐奖励一次，不重置任务。
        if = {
            limit = {
                mission_completed = a13_3_2
                gcd_crovis_in_restoration_war_trigger = yes
                NOT = { has_country_flag = gcd_crovis_task_reward_granted }
            }
            gcd_crovis_grant_task_reward_effect = yes
        }
        if = {
            limit = {
                has_country_modifier = gcd_crovis_funded_armies
                NOT = { any_country = { has_country_flag = gcd_crovis_actor is_in_war = { casus_belli = cb_crovis_invasion defender_leader = Z18 } } }
            }
            gcd_crovis_remove_support_effect = yes
        }
        # 剧情开战时与灾难存续期间均可检查；灾难结束后此倒戈检查自然失效。
        gcd_crovis_westmoors_hook_effect = yes
        if = {
            limit = { gcd_crovis_revolt_active_trigger = yes NOT = { has_country_flag = gcd_crovis_result_selected } }
            if = {
                limit = { gcd_crovis_crown_won_trigger = yes }
                set_country_flag = gcd_crovis_result_selected
                set_country_flag = gcd_crovis_crown_won
                country_event = { id = gawed_yongdongbianqu.8 days = 1 }
            }
            else_if = {
                limit = { gcd_crovis_rebels_won_trigger = yes }
                set_country_flag = gcd_crovis_result_selected
                country_event = { id = gawed_yongdongbianqu.7 days = 1 }
            }
        }
    }
}
