# 修正（内战后栈溢出）：停止使用会在事件链中重入的 .90 自我调度。
# 参考本机 feiten_daily_pulse，只在固定省份挂一个无数值、不可见的每日检查。
gcd_welyam_install_daily_pulse_effect = {
    if = {
        limit = { NOT = { has_country_flag = gcd_welyam_daily_pulse_installed } }
        set_country_flag = gcd_welyam_daily_pulse_installed
        221 = { add_province_triggered_modifier = gcd_welyam_daily_pulse }
    }
}

gcd_welyam_remove_daily_pulse_effect = {

    clr_country_flag = gcd_welyam_last_daily_tick
    clr_country_flag = gcd_welyam_daily_flip
    221 = {
        remove_province_triggered_modifier = gcd_welyam_daily_pulse
        clr_province_flag = gcd_welyam_daily_flip
    }
}

gcd_welyam_daily_reform_effect = {
    if = {
        limit = { has_disaster = gcd_welyam_reform }
        if = {
            limit = {
                NOT = { has_country_flag = gcd_welyam_civil_war_blocked }
                NOT = { has_country_flag = gcd_welyam_civil_war_happened }
                NOT = { has_country_flag = gcd_welyam_civil_war_queued }
                NOT = { estate_loyalty = { estate = estate_nobles loyalty = 20 } }
            }
            # 新增：仅首次低于 20 时记日期，持续低忠诚不能每天覆盖起算日期。
            if = {
                limit = { NOT = { has_country_flag = gcd_welyam_low_nobles_since } }
                set_country_flag = gcd_welyam_low_nobles_since
            }
            if = {
                limit = { had_country_flag = { flag = gcd_welyam_low_nobles_since days = 365 } }
                set_country_flag = gcd_welyam_civil_war_queued
                # 修正：将界面事件延迟一天，避免在每日回调栈中直接进入内战事件链。
                country_event = { id = gcd_welyam.100 days = 1 }
            }
        }
        else = {
            # 新增：忠诚恢复到 20（包括正好 20），连续一年计时立即作废。
            clr_country_flag = gcd_welyam_low_nobles_since
        }
        if = {
            limit = {
                NOT = { has_country_flag = gcd_welyam_rebel_spawn_stopped }
                had_country_flag = { flag = gcd_welyam_rebel_clock days = 730 }
            }
            # 新增：达到两年先挂待生成标记；没有空省时继续等待。
            set_country_flag = gcd_welyam_fixed_rebels_due
        }
        if = {
            limit = {
                has_country_flag = gcd_welyam_fixed_rebels_due
                NOT = { has_country_flag = gcd_welyam_rebel_spawn_stopped }
                NOT = { has_country_flag = gcd_welyam_rebel_event_queued }
                any_owned_province = { gcd_welyam_safe_rebel_province_trigger = yes }
            }
            # 修正（用户补充）：每两年投递一次专用隐藏事件；不在事件中自我调度。
            set_country_flag = gcd_welyam_rebel_event_queued
            # 以投递日作为下一轮起点，使两次隐藏事件间隔保持 730 天。
            clr_country_flag = gcd_welyam_rebel_clock
            set_country_flag = gcd_welyam_rebel_clock
            country_event = { id = gcd_welyam.91 days = 1 }
        }
    }
    gcd_welyam_check_civil_war_outcome_effect = yes
}

# 新增：只选择己方控制且没有部队的本国省份，不会调整已有叛军或删掉玩家部队。
# 暂无空省时保留到期标记，之后每日重试；任务二或灾难结束会撤销待生成标记。
gcd_welyam_try_spawn_fixed_rebels_effect = {
    random_owned_province = {
        limit = { gcd_welyam_safe_rebel_province_trigger = yes }
        gcd_welyam_spawn_fixed_rebels_in_province_effect = yes
    }
}

# 新增：省份作用域。size 是倍率，因此先生成贵族叛军，再按 REB 团数校准到恰好 12 团。
# 初始省份必须没有部队，脚本同步执行期间这里的 REB 全是本次新生成的贵族叛军。
gcd_welyam_spawn_fixed_rebels_in_province_effect = {
    set_variable = { gcd_welyam_rebel_seed_steps = 0 }
    set_variable = { gcd_welyam_rebel_trim_steps = 0 }
    while = {
        limit = {
            NOT = { num_of_units_in_province = { who = REB amount = 12 } }
            NOT = { check_variable = { gcd_welyam_rebel_seed_steps = 12 } }
        }
        spawn_rebels = { type = noble_rebels size = 3 }
        change_variable = { gcd_welyam_rebel_seed_steps = 1 }
    }
    while = {
        limit = {
            num_of_units_in_province = { who = REB amount = 13 }
            NOT = { check_variable = { gcd_welyam_rebel_trim_steps = 512 } }
        }
        # 本机 Anbennar 也使用 kill_units / who=REB / amount 来精确处理叛军团。
        kill_units = { who = REB amount = 1 }
        change_variable = { gcd_welyam_rebel_trim_steps = 1 }
    }
    if = {
        limit = {
            num_of_units_in_province = { who = REB amount = 12 }
            NOT = { num_of_units_in_province = { who = REB amount = 13 } }
        }
        owner = {
            clr_country_flag = gcd_welyam_fixed_rebels_due
            clr_country_flag = gcd_welyam_fixed_rebels_failed
            # 时钟由投递隐藏事件的灾难计时器维护，避免事件延迟的一天逐轮累计。
        }
    }
    else = {
        # 防护：若引擎未能校准成功，回收本次新叛军并保留重试，不留下错误规模。
        kill_units = { who = REB }
        owner = { set_country_flag = gcd_welyam_fixed_rebels_failed }
    }
    set_variable = { gcd_welyam_rebel_seed_steps = 0 }
    set_variable = { gcd_welyam_rebel_trim_steps = 0 }
}

        clr_country_flag = gcd_welyam_civil_war_happened

gcd_welyam_check_civil_war_outcome_effect = {
    if = {
        limit = {
            tag = A13
            gcd_welyam_civil_war_active_trigger = yes
            NOT = { has_country_flag = gcd_welyam_civil_war_result_delivered }
        }
        # 修正：旧版曾只设 queued，实际事件却未到达，导致永久禁止再次检查。
        # 迁移没有胜者标记的旧队列；新队列三天未收到事件时允许重投。
        if = {
            limit = {
                has_country_flag = gcd_welyam_civil_war_result_queued
                OR = {
                    AND = {
                        NOT = { has_country_flag = gcd_welyam_civil_war_lords_selected }
                        NOT = { has_country_flag = gcd_welyam_civil_war_crown_selected }
                    }
                    had_country_flag = { flag = gcd_welyam_civil_war_result_queued days = 3 }
                }
            }
            clr_country_flag = gcd_welyam_civil_war_result_queued
        }
        if = {
            limit = {
                NOT = { has_country_flag = gcd_welyam_civil_war_lords_selected }
                NOT = { has_country_flag = gcd_welyam_civil_war_crown_selected }
            }
            if = {
                limit = { gcd_welyam_capital_lost_trigger = yes }
                set_country_flag = gcd_welyam_civil_war_lords_selected
            }
            else_if = {
                limit = { war_score_against = { who = A50 value = 99 } }
                set_country_flag = gcd_welyam_civil_war_crown_selected
            }
        }
        if = {
            limit = { NOT = { has_country_flag = gcd_welyam_civil_war_result_queued } }
            if = {
                limit = { has_country_flag = gcd_welyam_civil_war_lords_selected }
                set_country_flag = gcd_welyam_civil_war_result_queued
                country_event = { id = gcd_welyam.101 days = 1 }
            }
            else_if = {
                limit = { has_country_flag = gcd_welyam_civil_war_crown_selected }
                set_country_flag = gcd_welyam_civil_war_result_queued
                country_event = { id = gcd_welyam.102 days = 1 }
            }
        }
    }
}
